cmake_minimum_required(VERSION 3.20)
project(Labs CXX)
set(CMAKE_CXX_STANDARD 20)

file(GLOB LAB_DIRS LIST_DIRECTORIES true ${CMAKE_SOURCE_DIR}/*)
foreach(dir ${LAB_DIRS})
    get_filename_component(name ${dir} NAME)
    if(IS_DIRECTORY ${dir} AND NOT name MATCHES "^(cmake-build|\\.)")
        file(GLOB_RECURSE SRCS CONFIGURE_DEPENDS ${dir}/*.cpp)
        if(SRCS)
            add_executable(${name} ${SRCS})
        endif()
    endif()
endforeach()
